Room 315 VLA Payload Case Runbook

The active Room 315 workflow is the curated 40 payload training cases. Legacy station-task orchestration has been removed from the runtime path.

Keep the golden dataset immutable when testing. Use fresh test output directories and only replace a golden dataset after the new run is fully approved.

Every New Terminal

cd /home/tiago/mfja_3rd_floor_ros2_ws
source /opt/ros/jazzy/setup.bash
source install/setup.bash

Optional Rebuild After Code Changes

cd /home/tiago/mfja_3rd_floor_ros2_ws
source /opt/ros/jazzy/setup.bash
colcon build --symlink-install --packages-select mfja_robot_control_config
source install/setup.bash

Manual Launch Room 315

ros2 launch mfja_3rd_floor_bringup room_315_only.launch.py \
  robots:=none \
  start_paused:=false \
  gui:=true \
  enable_room315_kinematic_shuttles:=true \
  enable_room315_vla:=true \
  enable_room315_vla_dataset_recorder:=true

Dry Run One Case

ros2 run mfja_robot_control_config room_315_pddl_scenario_generator.py \
  --case-id right_loaded_r1_s1_to_slot3_no_blocker \
  --language-template-id loaded_shuttle_to_slot \
  --dry-run

Preflight One Case

ros2 run mfja_robot_control_config room_315_pddl_scenario_generator.py \
  --case-id right_loaded_r1_s1_to_slot3_no_blocker \
  --language-template-id loaded_shuttle_to_slot \
  --command-timeout-s 8 \
  --preflight-only \
  --ready-line

Execute One Case

ros2 run mfja_robot_control_config room_315_pddl_scenario_generator.py \
  --case-id right_loaded_r1_s1_to_slot3_no_blocker \
  --language-template-id loaded_shuttle_to_slot \
  --command-timeout-s 30 \
  --arrival-timeout-s 120 \
  --require-dataset-recorder \
  --quiet \
  --execute

Test The 40 Golden Cases

Terminal 1 - Run The Full 40-Case Batch

cd /home/tiago/mfja_3rd_floor_ros2_ws
source /opt/ros/jazzy/setup.bash
source install/setup.bash

ros2 run mfja_robot_control_config room_315_payload_case_batch_runner.py \
  --case-config /home/tiago/mfja_3rd_floor_ros2_ws/src/mfja_3rd_floor_gz/mfja_robot_control_config/config/room_315_vla/payload_training_cases.yaml \
  --dataset-dir /home/tiago/room315_payload_cases_1_40_test \
  --results-dir /tmp/room315_payload_cases_1_40_test \
  --reset-dataset \
  --gui true \
  --launch-wait-s 15 \
  --preflight-timeout-s 90 \
  --command-timeout-s 45 \
  --arrival-timeout-s 240 \
  --episode-write-timeout-s 20

Terminal 2 - Check The 40-Case Result After Terminal 1 Finishes

cd /home/tiago/mfja_3rd_floor_ros2_ws
source /opt/ros/jazzy/setup.bash
source install/setup.bash

jq '{case_count, successes, failures, failed_cases}' \
  /tmp/room315_payload_cases_1_40_test/payload_case_batch_summary.json

jq '{approved_episodes, skipped_episodes, skipped_rows, rows, skip_reasons}' \
  /home/tiago/room315_payload_cases_1_40_test/meta/training_events.jsonl.summary.json

find /home/tiago/room315_payload_cases_1_40_test/episodes \
  -maxdepth 2 \
  -name validation.json \
  -print0 | xargs -0 jq -r \
  'select((.approved_for_training != true) or (.validation_status != "approved") or (.task_success != true)) | input_filename + " " + (.failure_reason // "")'
